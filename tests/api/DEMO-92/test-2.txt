package com.my.proj.resource;

import io.quarkus.test.junit.QuarkusTest;
import io.restassured.http.ContentType;
import jakarta.persistence.EntityManager;
import jakarta.transaction.Transactional;
import org.junit.jupiter.api.Test;
import static io.restassured.RestAssured.given;
import static org.hamcrest.CoreMatchers.is;
import static org.hamcrest.Matchers.not;

@QuarkusTest
public class Criterion2_PasswordHashed {

    @Transactional
    void setup() {
        User user = new User();
        user.name = "Jane Doe";
        user.email = "jane.doe@example.com";
        user.password = "securePassword123";

        given()
            .contentType(ContentType.JSON)
            .body(user)
            .when().post("/users");
    }

    @Test
    public void testPasswordHashed() {
        setup();

        User user = (User) EntityManager.createEntityManager()
                .createQuery("SELECT u FROM User u WHERE u.email = :email")
                .setParameter("email", "jane.doe@example.com")
                .getSingleResult();

        assert !user.password.equals("securePassword123");
    }
}
