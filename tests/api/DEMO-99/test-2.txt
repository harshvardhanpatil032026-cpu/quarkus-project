package com.my.proj.resource;

import io.quarkus.test.junit.QuarkusTest;
import io.restassured.http.ContentType;
import org.junit.jupiter.api.Test;

import java.util.Map;
import java.util.UUID;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.containsString;
import static org.hamcrest.Matchers.equalTo;
import static org.hamcrest.Matchers.hasItem;
import static org.hamcrest.Matchers.notNullValue;

@QuarkusTest
public class Criterion2_EmailIsTrimmedAndLowercased {

    private static final String OLD_PASSWORD = "OldP@ss1";
    private static final String NEW_PASSWORD = "NewP@ss2";

    /** Registers a user through POST /users and returns the email it was registered with. */
    private String registerUser(String email) {
        given().contentType(ContentType.JSON)
            .body(Map.of("name", "Reset Tester", "email", email, "password", OLD_PASSWORD))
            .when().post("/users")
            .then().statusCode(201);
        return email;
    }

    private String uniqueEmail() {
        return "reset-" + UUID.randomUUID() + "@example.com";
    }

    private io.restassured.response.ValidatableResponse reset(Object email, Object current, Object next) {
        java.util.Map<String, Object> body = new java.util.HashMap<>();
        body.put("email", email);
        body.put("currentPassword", current);
        body.put("newPassword", next);
        return given().contentType(ContentType.JSON).body(body)
            .when().post("/users/reset-password")
            .then();
    }

    @Test
    public void emailWithSurroundingSpacesAndUpperCaseIsMatched() {
        String email = registerUser(uniqueEmail());

        reset("  " + email.toUpperCase() + "  ", OLD_PASSWORD, NEW_PASSWORD)
            .statusCode(204);
    }

    @Test
    public void mixedCaseEmailIsMatched() {
        String email = registerUser(uniqueEmail());
        String mixed = email.substring(0, 1).toUpperCase() + email.substring(1);

        reset(mixed, OLD_PASSWORD, NEW_PASSWORD).statusCode(204);
    }
}
